# Vimium — Custom key mappings
#
# Спека: docs/keybinds.md §9, таблица: docs/keybinds-reference.md §11.
# Этот файл авторитетен, живой браузер — нет.
#
# УСТАНОВКА (руками, скриптом нельзя — Vimium держит бинды в storage
# расширения, а не в файле):
#   1. Vimium Options → Custom key mappings → вставить всё ниже
#   2. Там же включить filtered hints («Use link text for filtering»):
#      метки хинтов становятся цифрами, и `f` начинает работать в RU
#      раскладке тоже. Буквенные метки в RU не набираются в принципе,
#      дублями это не чинится (§9.5).
#
# Не работает на about:*, view-source:*, addons.mozilla.org и странице
# новой вкладки — WebExtensions туда не инжектятся.
#
# У части действий два входа: канон схемы и то, что уже наработано пальцами.
# Vimium держит и короткий бинд, и длинный с тем же префиксом, а команда
# может висеть на нескольких клавишах — так что оба живут рядом.


# ─── Снятые дефолты ──────────────────────────────────────────────────
# J/K — у Vimium это вкладки, а в схеме вкладки на H/L (§9.3).
# t/T/x — создание, поиск и закрытие вкладки одной голой буквой:
# слишком легко нажать случайно. Остаются <a-enter> и <a-q>.

unmap J
unmap K
unmap t
unmap T
unmap x


# ─── Вкладки ─────────────────────────────────────────────────────────
# Alt+1..9 = вкладка N делает сам Firefox, биндить нечего.

map H previousTab
map L nextTab
map <a-h> previousTab
map <a-l> nextTab
map <a-p> previousTab
map <a-n> nextTab
map <a-enter> createTab
map <a-q> removeTab


# ─── История ─────────────────────────────────────────────────────────
# Ctrl+O/Ctrl+I — канон схемы (§14): те же клавиши в nvim, zathura и
# Obsidian. Alt+j/k — второй вход; в браузере нет панелей, которыми Alt+jk
# управляет в остальных местах, так что клавиши свободны.

map <c-o> goBack
map <c-i> goForward
map <a-j> goBack
map <a-k> goForward


# ─── Omnibar (ярус Пробел, §9.4) ─────────────────────────────────────
# Одиночный пробел — быстрый вход; <space><space> оставлен, потому что
# именно он записан в схеме и жмётся одинаково во всех редакторах.

map <space> Vomnibar.activate
map <space><space> Vomnibar.activate
map <space>, Vomnibar.activateTabSelection
map <space>e Vomnibar.activateBookmarks
map <space>b Vomnibar.activateBookmarks
map <space>q removeTab


# ─── Прыжки по ссылкам ───────────────────────────────────────────────
# s/S — мышечная память flash из редактора. Дефолтные f/F остаются.

map s LinkHints.activateMode
map S LinkHints.activateModeToOpenInNewTab


# ─── Выделение ───────────────────────────────────────────────────────
# Alt+z — то же место, что copy-mode в tmux.

map <a-z> enterVisualMode


# ─── Кириллица (§10) ─────────────────────────────────────────────────
# langmap у Vimium нет, поэтому дубли по физической позиции клавиши.
# Хинты сюда не входят — их закрывает filtered mode, см. шапку.

# вкладки и история
map Р previousTab
map Д nextTab
map <a-р> previousTab
map <a-д> nextTab
map <a-з> previousTab
map <a-т> nextTab
map <a-й> removeTab
map <c-щ> goBack
map <c-ш> goForward
map <a-о> goBack
map <a-л> goForward

# omnibar
map <space>б Vomnibar.activateTabSelection
map <space>у Vomnibar.activateBookmarks
map <space>и Vomnibar.activateBookmarks
map <space>й removeTab

# выделение
map <a-я> enterVisualMode

# скролл
map о scrollDown
map л scrollUp
map в scrollPageDown
map г scrollPageUp
map <c-в> scrollPageDown
map <c-г> scrollPageUp
map пп scrollToTop
map П scrollToBottom

# прыжки по ссылкам
map а LinkHints.activateMode
map А LinkHints.activateModeToOpenInNewTab
map ы LinkHints.activateMode
map Ы LinkHints.activateModeToOpenInNewTab

# поиск по странице
# `.` — дубль `/` по физической позиции. Побочка: в EN-раскладке точка
# тоже начнёт поиск. Vimium её ничем не занимает, цена нулевая.
map . enterFindMode
map т performFind
map Т performBackwardsFind

# адреса и вкладки
map щ Vomnibar.activate
map Щ Vomnibar.activateInNewTab
map Е Vomnibar.activateTabSelection
map и Vomnibar.activateBookmarks
map И Vomnibar.activateBookmarksInNewTab
map Ч restoreTab
map нн copyCurrentUrl
map на LinkHints.activateModeToCopyLinkUrl
map к reload

# прочее
map ш enterInsertMode
map пш focusInput
map пг goUp
map хх goPrevious
map ъъ goNext

# Дубля на `?` (showHelp) нет намеренно: по физической позиции он дал бы
# `,`, а запятая читается в обеих раскладках — в EN она бы отбирала
# клавишу ради бинда, который жмут раз в месяц. В RU help = Shift+7.
#
# Нет дублей `е` (createTab) и `ч` (removeTab): голые t и x сняты выше,
# и RU-эквиваленты вернули бы ровно тот случайный промах, ради которого
# они снимались.
