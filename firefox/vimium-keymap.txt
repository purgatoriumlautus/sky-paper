# Vimium — Custom key mappings
#
# Спека: docs/keybinds.md §9, таблица: docs/keybinds-reference.md §11.
# Этот файл авторитетен, живой браузер — нет.
#
# УСТАНОВКА (руками, скриптом нельзя — Vimium держит бинды в storage
# расширения, а не в файле):
#   1. Vimium Options → Custom key mappings → вставить всё ниже
#   2. Там же, ниже по странице, включить «Ignore keyboard layout».
#
# Про «Ignore keyboard layout» — почему кириллических дублей здесь больше
# нет. Галка переключает Vimium с `event.key` на `event.code`
# (lib/keyboard_utils.js, getKeyChar): клавиша опознаётся по физической
# позиции, а не по букве, которую печатает раскладка. Один бинд `map о
# scrollDown` в EN-нотации ловится и в RU. Хинты идут через тот же путь,
# так что и `f` работает в RU — filtered hints для этого больше не нужны
# (как режим они по-прежнему годятся, но это уже вкус, а не костыль).
#
# Не работает на about:*, view-source:*, addons.mozilla.org и странице
# новой вкладки — WebExtensions туда не инжектятся. Там остаются только
# родные чорды Firefox, перебить их Vimium не может в принципе: они
# обрабатываются в chrome-слое, до контент-скрипта.
#
# У части действий два входа: канон схемы и то, что уже наработано пальцами.
# Vimium держит и короткий бинд, и длинный с тем же префиксом, а команда
# может висеть на нескольких клавишах — так что оба живут рядом.


# ─── Снятые дефолты ──────────────────────────────────────────────────
# J/K — у Vimium это вкладки, а в схеме вкладки на H/L (§9.3).
# t/T/x — создание, поиск и закрытие вкладки одной голой буквой:
# слишком легко нажать случайно. Жизненный цикл вкладки теперь вообще не
# здесь, а в about:keybinds — см. блок «Вкладки» ниже.
# Заглавная X (restoreTab) не снимается — промахнуться ею нечем, и это
# запасной вход на случай, если about:keybinds опять сломается.

unmap J
unmap K
unmap t
unmap T
unmap x


# ─── Вкладки ─────────────────────────────────────────────────────────
# Alt+1..9 = вкладка N делает сам Firefox, биндить нечего.
#
# Создание, закрытие и возврат вкладки здесь НЕ биндятся — они живут в
# about:keybinds (Firefox 153+, экспериментальный компонент customkeys,
# пишет ~/.config/mozilla/firefox/<профиль>/customKeys.json):
#
#   Alt+O        New Tab
#   Alt+Q        Close Tab
#   Alt+Shift+Q  Reopen Last Closed Tab
#
# Назначать в EN-раскладке: та страница пишет символ (key: "Q"), а не
# позицию, так что чорд из RU запишется как "Й" и отвалится в EN.
#
# Alt+O, а не Alt+Enter (как «создать» в остальной схеме), потому что
# about:keybinds не умеет кодировать Enter: он строит имя "VK_ENTER", а
# Gecko знает только VK_RETURN. Нераспознанное имя даёт keycode 0, а <key>
# с нулевым кодом совпадает по ОДНИМ модификаторам — после такой привязки
# любой Alt+клавиша открывает вкладку. Чинится только правкой
# customKeys.json руками, а он в профиле и на вторую машину не переносится.
#
# Почему там, а не тут: Vimium — контент-скрипт, он не инжектится в
# about:*, view-source:*, addons.mozilla.org и новую вкладку. То есть
# сразу после «открыть вкладку» бинд «закрыть вкладку» уже не работал бы.
# Родной XUL-<key> работает в окне всегда. Один чорд — один владелец,
# иначе они дерутся: Vimium перехватывает keydown первым и делает
# preventDefault, и до chrome-слоя событие не доходит.
#
# Переключение вкладок (H/L, Alt+h/l, Alt+p/n) остаётся здесь: нативного
# эквивалента для «предыдущая/следующая» в customkeys нет.

map H previousTab
map L nextTab
map <a-h> previousTab
map <a-l> nextTab
map <a-p> previousTab
map <a-n> nextTab


# ─── История ─────────────────────────────────────────────────────────
# Ctrl+O/Ctrl+I — канон схемы (§14): те же клавиши в nvim, zathura и
# Obsidian. Alt+j/k — второй вход; в браузере нет панелей, которыми Alt+jk
# управляет в остальных местах, так что клавиши свободны.

map <c-o> goBack
map <c-i> goForward
map <a-j> goBack
map <a-k> goForward


# ─── Omnibar (ярус Пробел, §9.4) ─────────────────────────────────────
# Одиночного `map <space> Vomnibar.activate` здесь намеренно нет.
# Vimium строит из биндов дерево (installKeyStateMapping в
# background_scripts/commands.js) и на узле, у которого уже есть своя
# команда, обрывает ветку — «existing command bindings take priority».
# Пробел-как-команда объявлялся первым, поэтому весь ярус <space>X ниже
# просто не устанавливался. Быстрый вход одним пробелом стоил всего
# яруса; ярус дороже.

map <space><space> Vomnibar.activate
map <space>, Vomnibar.activateTabSelection
map <space>e Vomnibar.activateBookmarks
map <space>b Vomnibar.activateBookmarks


# ─── Прыжки по ссылкам ───────────────────────────────────────────────
# s/S — мышечная память flash из редактора. Дефолтные f/F остаются.

map s LinkHints.activateMode
map S LinkHints.activateModeToOpenInNewTab


# ─── Выделение ───────────────────────────────────────────────────────
# Alt+z — то же место, что copy-mode в tmux.

map <a-z> enterVisualMode
